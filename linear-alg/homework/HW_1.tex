        Let $\textbf{A} = [a_{ij}]$ be a square matrix
        \begin{itemize}
            \item \textbf{A} is said to be a \textbf{\textit{symmetric matrix}} whenever $\textbf{A} = \textbf{A}^{T}$, i.e., whenever $a_{ij} = a_{ij}$
            \item \textbf{A} is called a \textbf{\textit{skew-symmetric}} whenever $\textbf{A} = -\textbf{A}^T$, i.e whenever $a_{ij} = -a_{ij}$
            \item \textbf{A} is said to be a \textbf{\textit{hermitian matrix}} whenever $\textbf{A} = \textbf{A}^*$, i.e., whenever $a_{ij} = \bar{a}_{ij}$.
            \item \textbf{A} is \textbf{\textit{skew hermitian}} when $\textbf{A} = -\textbf{A}^*$, i.e., whenever $a_{ij} = -\bar{a}_{ij}$.
        \end{itemize}
    \end{dfn}
